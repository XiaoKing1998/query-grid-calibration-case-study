def sha(path):
    d=hashlib.sha256()
    with path.open('rb') as f:
        for block in iter(lambda:f.read(8*1024**2),b''):d.update(block)
    return d.hexdigest()


def read(path):
    return json.loads(path.read_text(encoding='utf-8-sig'))


def readcsv(path):

def memory():
    class Counters(ctypes.Structure):
        _fields_=[('cb',ctypes.c_ulong),('PageFaultCount',ctypes.c_ulong),
                  ('PeakWorkingSetSize',ctypes.c_size_t),('WorkingSetSize',ctypes.c_size_t),
                  ('QuotaPeakPagedPoolUsage',ctypes.c_size_t),('QuotaPagedPoolUsage',ctypes.c_size_t),
                  ('QuotaPeakNonPagedPoolUsage',ctypes.c_size_t),('QuotaNonPagedPoolUsage',ctypes.c_size_t),
                  ('PagefileUsage',ctypes.c_size_t),('PeakPagefileUsage',ctypes.c_size_t)]
    kernel=ctypes.WinDLL('kernel32');kernel.GetCurrentProcess.restype=ctypes.c_void_p
    psapi=ctypes.WinDLL('psapi');psapi.GetProcessMemoryInfo.argtypes=[ctypes.c_void_p,ctypes.c_void_p,ctypes.c_ulong]
    c=Counters();c.cb=ctypes.sizeof(c)
    assert psapi.GetProcessMemoryInfo(kernel.GetCurrentProcess(),ctypes.byref(c),c.cb)
    return {'rss_bytes':int(c.WorkingSetSize),'process_peak_rss_bytes':int(c.PeakWorkingSetSize)}


def guard(plan):
    m=memory()
    assert time.perf_counter()-START<=plan['resources']['cpu_wall_seconds']
    assert m['process_peak_rss_bytes']<=plan['resources']['rss_bytes']
    assert sum(p.stat().st_size for p in OUT.rglob('*') if p.is_file())<=plan['resources']['output_bytes_limit']
    return m


def evaluate(scorebank,readout,thresholdbank,unit,threshold,groups,values,labels):
    records=[]
    for group,items in sorted(groups.items()):
        anomaly=any(labels[r['path']] for r in items)
        any_alarm=any(values[(scorebank,readout,r['path'])]>threshold for r in items)
        supported=any(values[(scorebank,readout,r['path'])]>threshold for r in items if labels[r['path']])
        records.append({'score_bank':scorebank,'readout':readout,'threshold_bank':thresholdbank,
                        'threshold_unit':unit,'threshold':threshold,'group_key':group,'is_anomaly':int(anomaly),
                        'any_alarm':int(any_alarm),'supported_alarm':int(supported)})
    normal=[r for r in records if not r['is_anomaly']]
    abnormal=[r for r in records if r['is_anomaly']]
    fp=sum(r['any_alarm'] for r in normal);anytp=sum(r['any_alarm'] for r in abnormal)
    support=sum(r['supported_alarm'] for r in abnormal)
    summary={'score_bank':scorebank,'readout':readout,'threshold_bank':thresholdbank,
             'threshold_unit':unit,'threshold':threshold,'normal_groups':len(normal),'abnormal_groups':len(abnormal),
             'normal_false_positive_groups':fp,'any_detected_abnormal_groups':anytp,
             'supported_detected_abnormal_groups':support,'normal_group_fpr':fp/len(normal),
             'any_group_tpr':anytp/len(abnormal),'supported_group_tpr':support/len(abnormal)}
    return summary,records
