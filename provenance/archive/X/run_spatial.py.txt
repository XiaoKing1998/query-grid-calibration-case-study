def sha(path):
    digest=hashlib.sha256()
    with path.open('rb') as f:
        for block in iter(lambda:f.read(8*1024**2), b''):
            digest.update(block)
    return digest.hexdigest()

def region_masks(np, n=64):
    rr,cc=np.indices((n,n))
    result={'top_left':(rr==0)|(cc==0),'bottom_right':(rr==n-1)|(cc==n-1),
            'outer_ring':(rr==0)|(cc==0)|(rr==n-1)|(cc==n-1),
            'interior':(rr>0)&(rr<n-1)&(cc>0)&(cc<n-1)}
    for r in (0,1):
        for c in (0,1):
            result[f'parity{r}{c}']=(rr%2==r)&(cc%2==c)
    return {k:v.ravel() for k,v in result.items()}


def summarize(np, row, method, distances, nearest, ties, source, masks):
    assert distances.shape==nearest.shape==ties.shape==(4096,)
    assert np.isfinite(distances).all() and (distances>=0).all()
    result={k:row[k] for k in IDENTITY}
    result['method']=method
    result.update(max=float(distances.max()),mean=float(distances.mean()),p95=float(np.quantile(distances,.95)))
    peak=int(np.argmax(distances))
    image_ties=distances==distances[peak]
    result['image_max_tie_count']=int(image_ties.sum())
    for name,mask in masks.items():
        result['image_max_ties_'+name]=int((image_ties&mask).sum())
    def location(prefix,index):
        proto=int(nearest[index]);origin=source[proto]
        result[prefix+'argmax_row']=index//64
        result[prefix+'argmax_col']=index%64
        result[prefix+'nearest_prototype']=proto
        result[prefix+'nearest_source_path']=origin['path']
        result[prefix+'nearest_source_row']=origin['row']
        result[prefix+'nearest_source_col']=origin['col']
        result[prefix+'nearest_source_camera']=origin['camera']
        result[prefix+'nearest_distance_tie_count']=int(ties[index])
    location('',peak)
    for name,mask in masks.items():
        allowed=np.flatnonzero(mask)
        index=int(allowed[np.argmax(distances[allowed])])
        result[name+'_max']=float(distances[index])
        result[name+'_max_tie_count']=int(np.count_nonzero(distances[allowed]==distances[index]))
        location(name+'_',index)
    return result


def match_blocks(torch, np, query_tensor, banks, target_ids, sham_ids, guard):
    """Identical FP32 arithmetic to upstream; shared balanced block for all subsets."""
    output={k:[[],[],[]] for k in METHODS}
    norms={k:(b*b).sum(1) for k,b in banks.items()}
    target=torch.as_tensor(target_ids,device=query_tensor.device,dtype=torch.long)
    sham=torch.as_tensor(sham_ids,device=query_tensor.device,dtype=torch.long)
    violation={'balanced_delete_topleft':0,'balanced_delete_interior_sham':0}
    def reduce(name,matrix):
        values,indices=matrix.min(1)
        tied=(matrix==values[:,None]).sum(1)
        for out,value in zip(output[name],(values,indices,tied)):
            out.append(value.cpu().numpy())
        return values
    with torch.no_grad():
        for q in query_tensor.split(256):
            guard()
            for name in ORIGINALS:
                bank=banks[name]
                d=(q*q).sum(1,keepdim=True)+norms[name][None]-2*(q@bank.T)
                d.clamp_min_(0)
                baseline=reduce(name,d)
                if name=='dino_balanced_own':
                    for method,deleted in (('balanced_delete_topleft',target),('balanced_delete_interior_sham',sham)):
                        masked=d.clone()
                        masked[:,deleted]=float('inf')
                        subset=reduce(method,masked)
                        bad=int((subset<baseline).sum().item())
                        violation[method]+=bad
                        assert bad==0
    return {k:tuple(np.concatenate(x) for x in v) for k,v in output.items()},violation

def safe_failure(error):
    value={'status':'STOP_SANITIZED_FAILURE','error_type':type(error).__name__,
           'stack':[{'file':Path(f.filename).name,'line':f.lineno,'function':f.name} for f in traceback.extract_tb(error.__traceback__)]}
    if not (OUT/'failure.json').exists():
        save('failure.json',value)
    print('WORKER_STOP',type(error).__name__,flush=True)
