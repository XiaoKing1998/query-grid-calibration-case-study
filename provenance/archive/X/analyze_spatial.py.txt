def check(name, value, detail=None):
    CHECKS.append({'check': name, 'pass': bool(value), 'detail': detail})
    if not value:
        raise ValueError('Independent check failed: ' + name)


def allowed_read(path):
    p = Path(path).resolve()
    return ((p.parent == OUT and p.name in ALLOWED_OUT)
            or (p.parent == UP.resolve() and p.name in ALLOWED_UP)
            or p in {INPUTS.resolve(), LABELS.resolve()})


def data(path):
    path = Path(path).resolve()
    if not allowed_read(path):
        raise ValueError('File outside independent verifier read allowlist')
    value = path.read_bytes()
    READS[str(path)] = {'sha256': hashlib.sha256(value).hexdigest(), 'bytes': len(value)}
    return value


def read_json(path):
    return json.loads(data(path).decode('utf-8-sig'))


def read_csv(path):
    return list(csv.DictReader(io.StringIO(data(path).decode('utf-8-sig'))))


def digest(path):
    path = Path(path).resolve()
    # Hash-only dependency guard: source/metadata under known research roots;
    # checkpoint paths must match the two established local weight filenames.
    roots = [REPO.resolve(), Path('${RESEARCH_WORKSPACE}').resolve()]
    checkpoint = path == Path('${USER_HOME}/.cache/torch/hub/checkpoints/dinov2_vitb14_pretrain.pth').resolve()
    permitted = checkpoint or (any(path.is_relative_to(r) for r in roots)
                               and (path.suffix.lower() in {'.py', '.json', '.csv', '.npy', '.zip', '.ps1'}
                                    or path.name == 'image_inputs.jsonl.gz'))
    known_vendor_python = path.is_relative_to((REPO / 'artifacts/ram_ad_20260918/vendor/dinov2_source/dinov2').resolve()) and path.suffix == '.py'
    if not permitted or (not known_vendor_python and any(x in path.name.lower() for x in ('token', 'credential', 'secret'))):
        raise ValueError('Unapproved dependency hash path')
    h = hashlib.sha256()
    size = 0
    with path.open('rb') as f:
        for block in iter(lambda: f.read(8 * 1024 * 1024), b''):
            h.update(block)
            size += len(block)
    READS[str(path)] = {'sha256': h.hexdigest(), 'bytes': size, 'hash_only': True}
    return h.hexdigest()

def compact_checks():
    families = collections.Counter()
    rolling = hashlib.sha256()
    for record in CHECKS:
        family = record['check'].split('_plastic_nut/')[0]
        families[family] += 1
        rolling.update(json.dumps(record, sort_keys=True, ensure_ascii=False, separators=(',', ':')).encode('utf-8') + b'\n')
    return {'checks_total': len(CHECKS), 'checks_failed': sum(not r['pass'] for r in CHECKS),
            'check_family_counts': dict(sorted(families.items())),
            'failed_checks': [r for r in CHECKS if not r['pass']],
            'ordered_full_check_records_sha256': rolling.hexdigest(),
            'persistence': 'All checks execute in memory; PASS records persist as counts and ordered-record digest to respect the 100 MiB budget.'}


def persistence_receipt(limit):
    path = OUT / 'analysis_persistence.json'
    files = [p for p in OUT.rglob('*') if p.is_file() and p != path]
    byte_count = sum(p.stat().st_size for p in files)
    report = {'status': 'PASS', 'other_persisted_files': len(files), 'other_persisted_bytes': byte_count,
              'limit_bytes': limit, 'including_this_receipt_bytes': byte_count}
    for _ in range(10):
        rendered = json.dumps(report, ensure_ascii=False, indent=2)
        total = byte_count + len(rendered.encode('utf-8'))
        if report['including_this_receipt_bytes'] == total:
            break
        report['including_this_receipt_bytes'] = total
    if total > limit:
        raise ValueError('Persisted byte limit exceeded after analysis outputs')
    path.write_bytes(rendered.encode('utf-8'))
    actual = sum(p.stat().st_size for p in OUT.rglob('*') if p.is_file())
    if actual != total or actual > limit:
        raise ValueError('Post-write persistence accounting mismatch')
    return report

def boolean(value):
    if value in (True, 1, '1', 'True', 'true'):
        return True
    if value in (False, 0, '0', 'False', 'false'):
        return False
    raise ValueError('Invalid boolean')


def threshold(values):
    values = np.asarray(values, float)
    rank = (95 * (len(values) + 1) + 99) // 100
    if rank > len(values):
        raise ValueError('Threshold rank outside finite observed sample')
    return float(np.sort(values)[rank - 1])


def group_rows(rows):
    groups = collections.defaultdict(list)
    for r in rows:
        groups[r['group_key']].append(r)
    groups = {k: sorted(v, key=lambda r: int(r['camera'])) for k, v in sorted(groups.items())}
    check('complete_five_camera_groups_' + str(len(rows)), all(
        len(v) == 5 and [int(r['camera']) for r in v] == [1, 2, 3, 4, 5]
        and len({r['research_role'] for r in v}) == 1 for v in groups.values()))
    return groups


def primary_arrays(full_target, full_sham, tl_target, tl_sham, in_target, in_sham):
    arrays = [np.asarray(x, float) for x in (full_target, full_sham, tl_target, tl_sham, in_target, in_sham)]
    if not all(a.shape == arrays[0].shape and np.isfinite(a).all() and (a > 0).all() for a in arrays):
        raise ValueError('Undefined primary log endpoint: nonpositive/nonfinite or mismatched arrays')
    a, b, c, d, e, f = arrays
    full = np.log(a / b)
    tl = np.log(c / d)
    interior = np.log(e / f)
    return {'full': full, 'top_left': tl, 'interior': interior, 'spatial_interaction': tl - interior}


def boot_summary(group_values, draws):
    group_values = np.asarray(group_values, float)
    means = group_values[draws].mean(axis=1)
    ci = np.quantile(means, [.025, .975])
    value = float(group_values.mean())
    return {'mean_log_contrast': value, 'conditional_percentile95': ci.tolist(),
            'geometric_ratio_or_ratio_of_ratios': float(np.exp(value)),
            'positive_lower_bound': bool(ci[0] > 0)}


def region_has(region, row, col):
    if region == 'top_left':
        return row == 0 or col == 0
    if region == 'bottom_right':
        return row == 63 or col == 63
    if region == 'outer_ring':
        return row in (0, 63) or col in (0, 63)
    if region == 'interior':
        return 1 <= row <= 62 and 1 <= col <= 62
    if region.startswith('parity'):
        return (row % 2, col % 2) == (int(region[-2]), int(region[-1]))
    raise ValueError('Unknown region')

def official_labels(query_rows):
    # Called only after successful terminal/freeze/control/integrity admission.
    with zipfile.ZipFile(io.BytesIO(data(LABELS))) as z:
        doc = json.loads(z.read('realiad_jsons/plastic_nut.json'))
    wanted = {r['path'] for r in query_rows}
    labels = {}
    for split in ('train', 'test'):
        for r in doc[split]:
            path = doc['meta']['prefix'] + r['image_path']
            if path in wanted:
                labels[path] = not (r['anomaly_class'] == doc['meta']['normal_class'] or r['mask_path'] is None)
    check('official_query_labels_complete', set(labels) == wanted)
    check('normal_calibration_official', all(not labels[r['path']] for r in query_rows if r['research_role'] == 'source_calibration'))
    check('258_development_abnormal_labels', sum(labels[r['path']] for r in query_rows if r['research_role'] == 'development') == 258)
    return labels

    group_values = [{'group_key': k, **{name: float(a[i].mean()) for name, a in arrays.items()}} for i, k in enumerate(cal)]
    cameras = []
    for camera in range(1, 6):
        for name, a in arrays.items():
            cameras.append({'camera': camera, 'endpoint': name, **boot_summary(a[:, camera - 1], draws)})
    gate = summaries['full']['positive_lower_bound'] and summaries['spatial_interaction']['positive_lower_bound']
    return {'unit': '76 five-view source-normal groups', 'seed': 20261012, 'bootstrap_samples': 5000,
            'endpoints': summaries, 'diagnostic_gate_pass': gate, 'innovation_proven': False,
            'per_camera': cameras}, group_values


def secondary_analysis(scalar, labels, oldfreeze):
    cal = group_rows([r for r in scalar if r['research_role'] == 'source_calibration'])
    dev = group_rows([r for r in scalar if r['research_role'] == 'development'])
    y = np.array([[labels[r['path']] for r in rr] for rr in dev.values()])
    normal, abnormal = y.sum(axis=1) == 0, y.sum(axis=1) > 0
    check('secondary_group_counts', int(normal.sum()) == 44 and int(abnormal.sum()) == 107)
    base_cal = np.array([[float(r[ORIGINAL]) for r in rr] for rr in cal.values()])
    base_tau = {'image': threshold(base_cal.ravel()), 'group': threshold(base_cal.max(axis=1))}
    check('old_balanced_image_threshold', base_tau['image'] == oldfreeze['image_thresholds'][ORIGINAL])
    check('old_balanced_group_threshold', base_tau['group'] == oldfreeze['group_thresholds'][ORIGINAL])
    summaries, predictions = [], []
    for m in METHODS:
        c = np.array([[float(r[m]) for r in rr] for rr in cal.values()])
        s = np.array([[float(r[m]) for r in rr] for rr in dev.values()])
        recal_tau = {'image': threshold(c.ravel()), 'group': threshold(c.max(axis=1))}
        for mode, taus in [('fixed_original_balanced', base_tau), ('independently_recalibrated', recal_tau)]:
            for unit, tau in taus.items():
                alarm = s > tau
                anyalarm, support = alarm.any(axis=1), (alarm & y).any(axis=1)
                summaries.append({'method': m, 'calibration': mode, 'threshold_unit': unit, 'threshold': tau,
                    'normal_groups': 44, 'abnormal_groups': 107,
                    'normal_false_positive_groups': int(anyalarm[normal].sum()),
                    'abnormal_any_detected_groups': int(anyalarm[abnormal].sum()),
                    'abnormal_supported_groups': int(support[abnormal].sum()),
                    'normal_group_fpr': float(anyalarm[normal].mean()),
                    'abnormal_any_group_tpr': float(anyalarm[abnormal].mean()),
                    'abnormal_supported_group_tpr': float(support[abnormal].mean())})
                for i, key in enumerate(dev):
                    predictions.append({'method': m, 'calibration': mode, 'threshold_unit': unit, 'threshold': tau,
                                        'group_key': key, 'is_anomaly': bool(abnormal[i]),
                                        'any_alarm': bool(anyalarm[i]), 'supported_alarm': bool(support[i])})
    return summaries, predictions


def migration_analysis(scalar, stat, usage, fit):
    reports = []
    for role in ('source_calibration', 'development'):
        paths = [r['path'] for r in scalar if r['research_role'] == role]
        for method in (METHODS[0], TARGET, SHAM):
            for region in [''] + REGIONS:
                pre = region + '_' if region else ''
                count = collections.Counter()
                for p in paths:
                    a, b = stat[(ORIGINAL, p)], stat[(method, p)]
                    ar, ac, br, bc = [int(r[k]) for r, k in [(a, pre + 'argmax_row'), (a, pre + 'argmax_col'), (b, pre + 'argmax_row'), (b, pre + 'argmax_col')]]
                    asp, bsp = a[pre + 'nearest_source_path'], b[pre + 'nearest_source_path']
                    count['representative_query_peak_changed'] += (ar, ac) != (br, bc)
                    count['representative_query_boundary_to_interior'] += region_has('outer_ring', ar, ac) and region_has('interior', br, bc)
                    count['representative_query_interior_to_boundary'] += region_has('interior', ar, ac) and region_has('outer_ring', br, bc)
                    count['representative_source_image_changed'] += asp != bsp
                    count['representative_source_group_changed'] += fit[asp]['group_key'] != fit[bsp]['group_key']
                    count['representative_source_camera_changed'] += fit[asp]['camera'] != fit[bsp]['camera']
                    count['representative_source_coordinate_changed'] += (asp, a[pre + 'nearest_source_row'], a[pre + 'nearest_source_col']) != (bsp, b[pre + 'nearest_source_row'], b[pre + 'nearest_source_col'])
                    peak_tie_key = region + '_max_tie_count' if region else 'image_max_tie_count'
                    unique_peaks = int(a[peak_tie_key]) == int(b[peak_tie_key]) == 1
                    unique_sources_at_unique_peaks = unique_peaks and int(a[pre + 'nearest_distance_tie_count']) == int(b[pre + 'nearest_distance_tie_count']) == 1
                    count['both_query_peaks_unique'] += unique_peaks
                    count['unique_query_peak_changed'] += unique_peaks and (ar, ac) != (br, bc)
                    count['both_query_peaks_and_nearest_sources_unique'] += unique_sources_at_unique_peaks
                    count['unique_source_group_changed'] += unique_sources_at_unique_peaks and fit[asp]['group_key'] != fit[bsp]['group_key']
                reports.append({'role': role, 'method': method, 'region': region or 'full', 'views': len(paths), **dict(count)})
    usage_summary = []
    for role in ('source_calibration', 'development'):
        paths = [r['path'] for r in scalar if r['research_role'] == role]
        for deletion in ('target', 'sham'):
            usage_summary.append({'role': role, 'deletion': deletion, 'views': len(paths),
                'query_cells': len(paths) * 4096,
                'representative_nn_in_deletion_cells': sum(int(usage[p][deletion + '_nn_cells']) for p in paths),
                'strictly_changed_cells': sum(int(usage[p][deletion + '_changed_cells']) for p in paths),
                'original_representative_argmax_uses_deletion_views': sum(boolean(usage[p]['argmax_uses_' + deletion]) for p in paths)})
    return reports, usage_summary


def tie_analysis(scalar, stat):
    records = []
    for role in ('source_calibration', 'development'):
        for method in METHODS:
            for camera in ('all', '1', '2', '3', '4', '5'):
                paths = [r['path'] for r in scalar if r['research_role'] == role and (camera == 'all' or r['camera'] == camera)]
                count = collections.Counter()
                for p in paths:
                    r = stat[(method, p)]
                    n = int(r['image_max_tie_count'])
                    tl = int(r['image_max_ties_top_left'])
                    inside = int(r['image_max_ties_interior'])
                    border = int(r['image_max_ties_outer_ring'])
                    other_border = border - tl
                    count['unique_image_peak_views'] += n == 1
                    count['multiple_image_peak_views'] += n > 1
                    count['all_image_peaks_top_left_views'] += tl == n
                    count['any_image_peak_top_left_views'] += tl > 0
                    count['all_image_peaks_interior_views'] += inside == n
                    count['any_image_peak_interior_views'] += inside > 0
                    count['all_image_peaks_other_boundary_views'] += other_border == n
                    count['any_image_peak_other_boundary_views'] += other_border > 0
                    count['all_image_peaks_outer_ring_views'] += border == n
                    count['mixed_border_interior_image_peaks_views'] += border > 0 and inside > 0
                    count['mixed_topleft_interior_otherboundary_peaks_views'] += sum(v > 0 for v in (tl, inside, other_border)) > 1
                    count['unique_nearest_source_at_representative_peak_views'] += int(r['nearest_distance_tie_count']) == 1
                    count['unique_peak_and_nearest_source_views'] += n == 1 and int(r['nearest_distance_tie_count']) == 1
                    for parity in ('parity00', 'parity01', 'parity10', 'parity11'):
                        count['all_image_peaks_' + parity + '_views'] += int(r['image_max_ties_' + parity]) == n
                        count['any_image_peak_' + parity + '_views'] += int(r['image_max_ties_' + parity]) > 0
                assert sum(count[k] for k in ('all_image_peaks_top_left_views', 'all_image_peaks_interior_views',
                           'all_image_peaks_other_boundary_views', 'mixed_topleft_interior_otherboundary_peaks_views')) == len(paths)
                records.append({'role': role, 'method': method, 'camera': camera, 'views': len(paths), **dict(count)})
    return records

def self_test():
    a = np.full((76, 5), 2.0)
    b = np.ones((76, 5))
    values = primary_arrays(a, b, a * 2, b, a, b)
    draws = np.random.Generator(np.random.PCG64(20261012)).integers(0, 76, (5000, 76))
    assert abs(boot_summary(values['full'].mean(1), draws)['mean_log_contrast'] - math.log(2)) < 1e-14
    assert abs(boot_summary(values['spatial_interaction'].mean(1), draws)['mean_log_contrast'] - math.log(2)) < 1e-14
    assert boot_summary(values['full'].mean(1), draws)['positive_lower_bound']
    zero = primary_arrays(b, b, b, b, b, b)
    assert not boot_summary(zero['full'].mean(1), draws)['positive_lower_bound']
    failed = False
    try:
        primary_arrays(a * 0, b, a, b, a, b)
    except ValueError:
        failed = True
    assert failed
    assert threshold(range(1, 381)) == 362 and threshold(range(1, 77)) == 74
    assert region_has('top_left', 0, 63) and region_has('bottom_right', 0, 63)
    assert not region_has('interior', 0, 63) and region_has('interior', 1, 62)
    assert region_has('parity01', 0, 63)
    result = {'status': 'SYNTHETIC_SELF_TEST_PASS', 'research_data_read': False,
              'checks': ['positive_log_effect', 'paired_interaction', 'strict_positive_gate', 'zero_endpoint_rejected',
                         'finite_sample_threshold_ranks', 'overlapping_edges_and_interior', 'parity_membership']}
    save('analysis_self_test.json', result)
    print(json.dumps(result))


def run():
    start = time.perf_counter()
    required = ['plan.json', 'preflight.json', 'intervention_identities.json', 'code_receipt.json', 'prediction_freeze.json',
                'spatial_statistics.csv', 'balanced_usage.csv', 'scalar_predictions.csv', 'supervisor.json',
                'plastic_nut_image_receipt.json', 'execution_monitor.json', 'launch_record.json']
    missing = [n for n in required if not (OUT / n).exists()]
    if missing:
        print(json.dumps({'status': 'WAIT_COMPLETE_INPUTS', 'missing': missing}))
        return 2
    try:
        plan = read_json(OUT / 'plan.json')
        preflight, freeze, runtime = verify_runtime(plan)
        inputs, scalar, stat, usage, fit, oldfreeze = verify_data(freeze)
        labels = official_labels(scalar)
        primary, primary_groups = primary_analysis(scalar, stat)
        secondary, predictions = secondary_analysis(scalar, labels, oldfreeze)
        migration, usage_summary = migration_analysis(scalar, stat, usage, fit)
        tie_summary = tie_analysis(scalar, stat)
        result = {'status': 'INDEPENDENT_SPATIAL_ANALYSIS_PASS', 'scientific_diagnostic_status': 'SUPPORTED' if primary['diagnostic_gate_pass'] else 'NOT_ESTABLISHED',
            'innovation_proven': False, 'command': [sys.executable, '-B', str(Path(__file__).resolve())],
            'checks_total': len(CHECKS), 'checks_failed': 0, 'primary': primary, 'secondary': secondary,
            'migration': migration, 'usage_summary': usage_summary, 'tie_summary': tie_summary, 'runtime': runtime, 'inputs': READS,
            'wall_seconds': time.perf_counter() - start,
            'limits': ['conditional on fitted bank and single matched sham', 'necessity does not identify mediation of odd-to-balanced change',
                       'source camera/parity matched, source image/group and feature distances not matched',
                       'no mask data; representative peak/source migration is not unique-location evidence',
                       'reused development; no confirmation or deployment-risk conclusion']}
        save('analysis_checks.json', compact_checks())
        save('analysis_result.json', result)
        csvsave('analysis_primary_groups.csv', primary_groups)
        csvsave('analysis_secondary.csv', secondary)
        csvsave('analysis_group_predictions.csv', predictions)
        csvsave('analysis_migration.csv', migration)
        csvsave('analysis_usage_summary.csv', usage_summary)
        csvsave('analysis_tie_summary.csv', tie_summary)
        write_report(result)
        persistence_receipt(plan['resources']['persisted_bytes_limit'])
        print(json.dumps({'status': result['status'], 'checks_total': len(CHECKS), 'primary': primary, 'wall_seconds': result['wall_seconds']}, ensure_ascii=False, indent=2))
        return 0
    except Exception as exc:
        save('analysis_checks_failed.json', compact_checks())
        save('analysis_failure.json', {'status': 'INDEPENDENT_ADMISSION_OR_ANALYSIS_FAILED',
              'error_type': type(exc).__name__, 'message': str(exc), 'checks_total': len(CHECKS),
              'checks_failed': sum(not c['pass'] for c in CHECKS), 'inputs': READS,
              'scientific_results_admitted': False})
        raise


if __name__ == '__main__':
