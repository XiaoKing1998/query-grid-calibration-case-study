"""Controlled DINO896 transfer of three external candidate rules.

Only --run after root-owned plan/admission freeze can request study pixels.
The public reservoir and prepool identity rules are transferred. The frozen
DINO896 embedding, 4096 bank and common approximate coreset are deliberate
controls, so this is not a native public PatchCore reproduction.
"""

sys.dont_write_bytecode = True
HERE = Path(__file__).resolve().parent
OUT = HERE.parent
U = Path('${RESEARCH_WORKSPACE}/contribution_upgrade_20261003/p1')
import candidate_plan as cp


def read(path):
    return json.loads(Path(path).read_text(encoding='utf-8-sig'))


def rows(path):
    with Path(path).open(encoding='utf-8-sig', newline='') as stream:
        return list(csv.DictReader(stream))


def save(name, value):
    cp.write_json(HERE/name, value)


def csvsave(name, value):
    cp.write_csv(HERE/name, value)


def module_at(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def check_admission():
    admission = read(OUT/'admission.json')
    plan = read(OUT/'plan.json')
    assert admission['status'] == 'PASS_METADATA_AND_LOCAL_DEPENDENCY_ADMISSION'
    assert cp.sha(OUT/'plan.json') == admission['plan_sha256']
    assert cp.sha(__file__) == admission['producer_sha256']
    bound = {**admission['dependency_hashes'], **admission['checkpoint_hashes']}
    for path, expected in bound.items():
        assert cp.sha(path) == expected
    required = [HERE/'candidate_plan.py', HERE/'candidate_metadata.json',
                HERE/'candidate_identities.csv', HERE/'candidate_trace.json',
                HERE/'source_orders.csv', U/'runner_draft.py',
                cp.OLD/'static_runner.py', cp.OLD/'run_matrix.py', cp.OLD/'transport.py',
                cp.OLD/'audit_archives.py', cp.OLD/'directories/plastic_nut/directory.json.gz',
                cp.OLD/'attempt02/results/plastic_nut/inputs.csv',
                OUT/'source_audit/enhanced_features.py', OUT/'source_audit/build_memory_bank.py']
    normalized = {str(Path(path).resolve()).casefold() for path in bound}
    assert all(str(path.resolve()).casefold() in normalized for path in required)
    assert plan['methods'] == list(cp.METHODS) and plan['input_roles'] == cp.COUNTS
    assert plan['bank_size'] == 4096
    assert plan['confirmation_pixels_and_scores'] == plan['masks_requested'] == 0
    limits = plan['resources']
    assert limits['gpu_wall_seconds'] == 900 and limits['supervisor_seconds'] == 1800
    assert limits['cpu_other_seconds'] == 600 and limits['rss_bytes'] == 20*1024**3
    assert limits['persisted_bytes_limit'] == 100*1024**2
    assert 0 < limits['image_acquisition_seconds'] <= limits['supervisor_seconds']
    prior = admission['prior_resource_use']
    assert 0 <= prior['execution_wall_seconds'] < 1800
    assert 0 <= prior['gpu_wall_seconds'] < 900
    meta = read(HERE/'candidate_metadata.json')
    assert meta['status'] == 'METADATA_CANDIDATE_IDENTITIES_COMPLETE_NO_PIXELS'
    assert meta['candidate_count_each'] == 20000
    assert meta['candidate_module_sha256'] == cp.sha(HERE/'candidate_plan.py')
    for filename, expected in meta['sha256_files'].items():
        assert cp.sha(HERE/filename) == expected
    return plan, admission, meta


def candidate_inputs(np, fit, meta):
    """Load the already frozen scalar identities; do not resample in inference."""
    pools = {name: np.empty(cp.CANDIDATES, dtype=np.int64) for name in cp.CONSTRUCTIONS}
    counts = {name: 0 for name in cp.CONSTRUCTIONS}
    for row in rows(HERE/'candidate_identities.csv'):
        name = row['construction']
        k = int(row['candidate_index'])
        assert name in pools and k == counts[name]
        i, cell = divmod(int(row['global_patch_id']), 4096)
        assert i == int(row['source_index']) and 0 <= i < 875
        assert row['path'] == fit[i]['path'] and row['group_key'] == fit[i]['group_key']
        assert str(row['camera']) == str(fit[i]['camera'])
        assert cell == int(row['row'])*64+int(row['col'])
        pools[name][k] = int(row['global_patch_id'])
        counts[name] += 1
    for name, ids in pools.items():
        assert counts[name] == 20000 and len(np.unique(ids)) == 20000
        assert hashlib.sha256(np.asarray(ids, dtype='<i8').tobytes()).hexdigest() == meta['ordered_candidate_sha256_le_i8'][name]
    return pools


def acquire_images(legacy, static, inputs, deadline):
    helper = module_at('external_sampling_transport', legacy.HELPER)
    helper.OUT = HERE
    events = []
    token = helper.local_token()
    try:
        for attempt in range(3):
            try:
                ram = helper.acquire('plastic_nut', inputs, token, deadline, 'image')
                events.append({'attempt': attempt+1, 'status': 'PASS'})
                return ram
            except helper.transport.RangeError as error:
                transient = str(error) == 'transport failure; no credentials logged'
                events.append({'attempt': attempt+1, 'status': 'TRANSIENT' if transient else 'VALIDATION_FAILURE',
                               'partial_response_bytes': 'UNKNOWN'})
                if not transient or attempt == 2 or time.perf_counter() >= deadline:
                    raise
    finally:
        del token
        save('transport_attempts.json', events)


def worker(nonce):
    started = time.perf_counter()
    claim = read(HERE/'execution_claim.json')
    assert claim['nonce'] == nonce
    plan, admission, meta = check_admission()
    import numpy as np
    from PIL import Image
    legacy = module_at('external_sampling_p1_library', U/'runner_draft.py')
    static = legacy.load_static()
    roles = static.load_manifest()['plastic_nut']
    assert {name: len(value) for name, value in roles.items()} == cp.COUNTS
    assert static.order_hash(roles['source_fit']) == meta['source_order_sha256']
    assert static.order_hash(roles['source_fit']) == claim['source_order_sha256']
    inputs = {row['path']: row for row in rows(cp.OLD/'attempt02/results/plastic_nut/inputs.csv')}
    assert len(inputs) == 2010 and set(inputs) == {row['path'] for role in static.RUN_ROLES for row in roles[role]}
    fit = roles['source_fit']
    pool_ids = candidate_inputs(np, fit, meta)
    take = {name: [np.flatnonzero(ids//4096 == i) for i in range(875)] for name, ids in pool_ids.items()}
    prior = admission['prior_resource_use']
    limits = plan['resources']
    torch = None
    gpu_started = None
    peak_cuda = None
    host_memory = None
    outcome = 'STOPPED_BEFORE_COMPLETION'
    ram = {}
    bank_hashes = {}
    scores, peaks, normal_statistics, bank_ids = [], [], [], []
    timings = {'acquisition_seconds': 0., 'decode_seconds': 0., 'fit_embed_seconds': 0.,
               'query_embed_seconds': 0., 'candidate_copy_seconds': 0., 'coreset_seconds': {},
               'nearest_seconds': {method: 0. for method in cp.METHODS}, 'summary_seconds': 0.}
    environment = None
    with static.Budget(claim['execution_deadline_perf_counter'], limits['rss_bytes'], static.WindowsHostMemory()) as whole_budget:
        try:
            whole_budget.check()
            begin = time.perf_counter()
            acquisition_deadline = min(claim['execution_deadline_perf_counter'], begin+limits['image_acquisition_seconds'])
            ram = acquire_images(legacy, static, inputs, acquisition_deadline)
            timings['acquisition_seconds'] = time.perf_counter()-begin
            assert time.perf_counter() <= acquisition_deadline
            whole_budget.check()
            gpu_started = time.perf_counter()
            compute_deadline = min(claim['execution_deadline_perf_counter'], gpu_started+limits['gpu_wall_seconds']-prior['gpu_wall_seconds'])
            with static.Budget(compute_deadline, limits['rss_bytes'], static.WindowsHostMemory()) as compute_budget:
                base = legacy.load_core()
                torch = base.torch
                torch.cuda.reset_peak_memory_stats()
                environment = {'python': sys.version, 'python_executable': sys.executable,
                               'numpy': np.__version__, 'torch': torch.__version__,
                               'cuda_runtime': torch.version.cuda, 'cuda_device': torch.cuda.get_device_name(0),
                               'threads': torch.get_num_threads(), 'allow_tf32': torch.backends.cuda.matmul.allow_tf32,
                               'cudnn_allow_tf32': torch.backends.cudnn.allow_tf32,
                               'matmul_precision': torch.get_float32_matmul_precision(),
                               'coreset_implementation': 'Pinned static_core.RecordingSampler ApproximateGreedy, projection128/anchors10',
                               'embedding': 'DINOv2 ViT-B14 layer11 normalized tokens, FP32, full896x896 ->64x64x768'}
                model = legacy.load_dino(torch)
                means = torch.tensor([.485, .456, .406], device='cuda')[:, None, None]
                stds = torch.tensor([.229, .224, .225], device='cuda')[:, None, None]

                def embed(row, phase):
                    whole_budget.check()
                    compute_budget.check()
                    static.assert_fetch_role(row)
                    assert row['path'] in ram
                    begin = time.perf_counter()
                    with Image.open(io.BytesIO(ram[row['path']])) as im:
                        assert im.size == (1024, 1024)
                        image = np.array(im.convert('RGB').resize((896, 896), Image.Resampling.BILINEAR), dtype=np.float32)/255.
                    timings['decode_seconds'] += time.perf_counter()-begin
                    begin = time.perf_counter()
                    x = torch.from_numpy(image.transpose(2, 0, 1)).cuda()
                    x = (x-means)/stds
                    with torch.no_grad():
                        feature = model.get_intermediate_layers(x[None], n=(11,), norm=True)[0][0].cpu().numpy()
                    timings[phase+'_embed_seconds'] += time.perf_counter()-begin
                    assert feature.shape == (4096, 768) and feature.dtype == np.float32 and np.isfinite(feature).all()
                    return feature

                pools = {name: np.empty((20000, 768), dtype=np.float32) for name in cp.CONSTRUCTIONS}
                filled = {name: 0 for name in cp.CONSTRUCTIONS}
                for i, row in enumerate(fit):
                    feature = embed(row, 'fit')
                    begin = time.perf_counter()
                    for name in cp.CONSTRUCTIONS:
                        destination = take[name][i]
                        pools[name][destination] = feature[pool_ids[name][destination] % 4096]
                        filled[name] += len(destination)
                    timings['candidate_copy_seconds'] += time.perf_counter()-begin
                    del ram[row['path']]
                    if (i+1) % 100 == 0:
                        print('FIT_DINO896', i+1, flush=True)
                assert filled == {name: 20000 for name in cp.CONSTRUCTIONS}
                banks = {}
                for name in cp.CONSTRUCTIONS:
                    for seed in cp.SEEDS:
                        whole_budget.check()
                        compute_budget.check()
                        method = f'{name}_s{seed}'
                        begin = time.perf_counter()
                        bank, indices = legacy.select_seeded(base, np, pools[name], seed, compute_budget.deadline)
                        torch.cuda.synchronize()
                        timings['coreset_seconds'][method] = time.perf_counter()-begin
                        assert bank.shape == (4096, 768) and bank.dtype == np.float32 and np.isfinite(bank).all()
                        indices = np.asarray(indices, dtype=np.int64)
                        assert indices.shape == (4096,) and len(np.unique(indices)) == 4096
                        bank_hashes[method] = hashlib.sha256(bank.tobytes(order='C')).hexdigest()
                        for prototype, k in enumerate(indices):
                            global_id = int(pool_ids[name][int(k)])
                            i, cell = divmod(global_id, 4096)
                            bank_ids.append({'method': method, 'construction': name, 'seed': seed,
                                             'prototype': prototype, 'candidate_index': int(k),
                                             'global_patch_id': global_id, 'source_index': i,
                                             'path': fit[i]['path'], 'group_key': fit[i]['group_key'],
                                             'camera': fit[i]['camera'], 'row': cell//64, 'col': cell%64})
                        banks[method] = torch.from_numpy(bank.copy()).cuda()
                        print('BANK_READY', method, flush=True)
                    del pools[name]
                    gc.collect()
                del pools, bank, indices, feature
                gc.collect()
                torch.cuda.empty_cache()
                for i, row in enumerate(roles['source_calibration']+roles['development']):
                    feature = embed(row, 'query')
                    record = {key: row[key] for key in legacy.IDENTITY}
                    for method in cp.METHODS:
                        whole_budget.check()
                        compute_budget.check()
                        begin = time.perf_counter()
                        distances = base.nearest_squared(feature, banks[method])
                        timings['nearest_seconds'][method] += time.perf_counter()-begin
                        assert distances.shape == (4096,) and np.isfinite(distances).all() and (distances >= 0).all()
                        begin = time.perf_counter()
                        peak = int(np.argmax(distances))
                        score = float(distances[peak])
                        record[method] = score
                        peaks.append({**{key: row[key] for key in legacy.IDENTITY}, 'method': method,
                                      'max': score, 'argmax_row': peak//64, 'argmax_col': peak%64,
                                      'image_max_tie_count': int(np.count_nonzero(distances == distances[peak])),
                                      'evaluated_query_cells': 4096})
                        if row['research_role'] == 'source_calibration':
                            normal_statistics.append({**{key: row[key] for key in legacy.IDENTITY},
                                'method': method, 'mean_squared_nearest': float(distances.mean()),
                                'p95_squared_nearest': float(np.quantile(distances, .95)),
                                'max_squared_nearest': score, 'evaluated_query_cells': 4096})
                        timings['summary_seconds'] += time.perf_counter()-begin
                    scores.append(record)
                    del ram[row['path']]
                    if (i+1) % 100 == 0:
                        print('QUERY_DINO896', i+1, flush=True)
                compute_budget.check()
                torch.cuda.synchronize()
                peak_cuda = int(torch.cuda.max_memory_allocated())
                del banks, model, feature, distances, means, stds
                gc.collect()
                torch.cuda.empty_cache()
            assert len(scores) == 1135 and len(peaks) == 10215 and len(normal_statistics) == 3420 and len(bank_ids) == 36864 and not ram
            for name, data in [('scalar_predictions.csv', scores), ('peak_statistics.csv', peaks),
                               ('normal_statistics.csv', normal_statistics), ('bank_identities.csv', bank_ids)]:
                csvsave(name, data)
            whole_budget.check()
            assert sum(path.stat().st_size for path in OUT.rglob('*') if path.is_file()) <= limits['persisted_bytes_limit']
            assert time.perf_counter()-gpu_started+prior['gpu_wall_seconds'] <= limits['gpu_wall_seconds']
            outcome = 'EXTERNAL_SAMPLING_PREDICTIONS_FROZEN'
        finally:
            gpu_elapsed = 0. if gpu_started is None else time.perf_counter()-gpu_started
            if torch is not None and torch.cuda.is_initialized():
                peak_cuda = int(torch.cuda.max_memory_allocated())
            try:
                host_memory = whole_budget.memory_report()
            except static.ResourceStop:
                # Keep real native counters even when the RSS gate itself failed.
                memory = whole_budget.process.memory_info()
                host_memory = {'rss_bytes': memory.rss,
                    'native_process_peak_rss_bytes': memory.native_process_peak_rss,
                    'private_bytes': memory.private, 'pagefile_bytes': memory.pagefile,
                    'memory_gate_failed': True}
            ram.clear()
            save('resource_receipt.json', {'status': outcome, 'gpu_wall_seconds': gpu_elapsed,
                 'prior_gpu_wall_seconds': prior['gpu_wall_seconds'],
                 'total_gpu_wall_seconds': prior['gpu_wall_seconds']+gpu_elapsed,
                 'worker_wall_seconds': time.perf_counter()-started,
                 'prior_execution_wall_seconds': prior['execution_wall_seconds'],
                 'gpu_wall_definition': 'One conservative compute-stage elapsed interval from backend/model load through output persistence; acquisition precedes it and counts only execution wall.',
                 'peak_cuda_allocated_bytes': peak_cuda, 'host_memory': host_memory, 'timings': timings,
                 'completed_query_images': len(scores), 'completed_banks': list(bank_hashes),
                 'confirmation_access': 0, 'masks_requested': 0, 'payload_saved': False})
    assert outcome == 'EXTERNAL_SAMPLING_PREDICTIONS_FROZEN'
    assert gpu_elapsed+prior['gpu_wall_seconds'] <= limits['gpu_wall_seconds']
    freeze = {'status': outcome, 'methods': cp.METHODS, 'counts': cp.COUNTS, 'bank_hashes': bank_hashes,
              'query_rows': 1135, 'query_cells': 4096, 'fit_embedded_images': 875,
              'candidate_count_each': 20000, 'candidate_metadata_sha256': cp.sha(HERE/'candidate_metadata.json'),
              'environment': environment, 'plan_sha256': cp.sha(OUT/'plan.json'),
              'admission_sha256': cp.sha(OUT/'admission.json'), 'producer_sha256': cp.sha(__file__),
              'sha256_files': {name: cp.sha(HERE/name) for name in ('scalar_predictions.csv', 'peak_statistics.csv', 'normal_statistics.csv', 'bank_identities.csv', 'resource_receipt.json', 'plastic_nut_image_receipt.json', 'transport_attempts.json')},
              'before_development_labels': True, 'confirmation_access': 0, 'masks_requested': 0,
              'payload_saved': False, 'native_public_patchcore_reproduction': False,
              'uniform_control': 'Direct uniform20k over complete fit history has the same distribution as uniform-history50k then uniform20k, not identical actual candidate sampling CPU cost.',
              'full_candidate_reference': 'Not included by prospective design; this study does not identify additional coreset approximation error.',
              'tie_policy': 'First row-major maximal query cell plus complete tie count; location representative does not imply localization correctness.'}
    save('prediction_freeze.json', freeze)
    print('EXTERNAL_SAMPLING_PREDICTIONS_FROZEN', flush=True)


def safe_failure(error):
    if not (HERE/'failure.json').exists():
        save('failure.json', {'status': 'STOP_SANITIZED_FAILURE', 'error_type': type(error).__name__,
                             'stack': [{'file': Path(frame.filename).name, 'line': frame.lineno, 'function': frame.name}
                                       for frame in traceback.extract_tb(error.__traceback__)]})
    print('WORKER_STOP', type(error).__name__, flush=True)


def main():
    if len(sys.argv) == 3 and sys.argv[1] == '--worker':
        worker(sys.argv[2])
        return 0
    assert sys.argv[1:] == ['--run']
    started = time.perf_counter()
    plan, admission, meta = check_admission()
    assert not (HERE/'execution_claim.json').exists()
    prior = admission['prior_resource_use']
    deadline = started+plan['resources']['supervisor_seconds']-prior['execution_wall_seconds']
    nonce = uuid.uuid4().hex
    save('execution_claim.json', {'nonce': nonce, 'parent_pid': os.getpid(),
         'producer_sha256': cp.sha(__file__), 'plan_sha256': cp.sha(OUT/'plan.json'),
         'source_order_sha256': meta['source_order_sha256'], 'execution_deadline_perf_counter': deadline})
    exit_code = None
    timeout = False
    try:
        result = subprocess.run([sys.executable, '-B', str(Path(__file__)), '--worker', nonce], timeout=max(.01, deadline-time.perf_counter()))
        exit_code = result.returncode
        return exit_code
    except subprocess.TimeoutExpired:
        timeout = True
        return 1
    finally:
        wall = time.perf_counter()-started
        save('supervisor.json', {'exit_code': exit_code, 'timeout': timeout,
             'execution_wall_seconds': wall, 'prior_execution_wall_seconds': prior['execution_wall_seconds'],
             'total_execution_wall_seconds': prior['execution_wall_seconds']+wall,
             'cpu_other_budget_separate': True, 'limit_execution_wall_seconds': 1800})


if __name__ == '__main__':
    try:
        sys.exit(main())
    except BaseException as error:
        if isinstance(error, SystemExit):
            raise
        safe_failure(error)
        sys.exit(1)
