def read_json(path):
    return json.loads(path.read_text(encoding='utf-8-sig'))


def read_csv(path):
    with path.open(encoding='utf-8-sig', newline='') as stream:
        return list(csv.DictReader(stream))


def sha256(path):
    digest = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(8 * 1024**2), b''):
            digest.update(block)
    return digest.hexdigest()


def check(name, value):

def peak_rss_bytes():
    if sys.platform != 'win32':
        return None
    class Counters(ctypes.Structure):
        _fields_ = [('cb', ctypes.c_ulong), ('PageFaultCount', ctypes.c_ulong),
                    ('PeakWorkingSetSize', ctypes.c_size_t), ('WorkingSetSize', ctypes.c_size_t),
                    ('QuotaPeakPagedPoolUsage', ctypes.c_size_t), ('QuotaPagedPoolUsage', ctypes.c_size_t),
                    ('QuotaPeakNonPagedPoolUsage', ctypes.c_size_t), ('QuotaNonPagedPoolUsage', ctypes.c_size_t),
                    ('PagefileUsage', ctypes.c_size_t), ('PeakPagefileUsage', ctypes.c_size_t)]
    kernel = ctypes.WinDLL('kernel32')
    kernel.GetCurrentProcess.restype = ctypes.c_void_p
    psapi = ctypes.WinDLL('psapi')
    psapi.GetProcessMemoryInfo.argtypes = [ctypes.c_void_p, ctypes.c_void_p, ctypes.c_ulong]
    counters = Counters()
    counters.cb = ctypes.sizeof(counters)
    if not psapi.GetProcessMemoryInfo(kernel.GetCurrentProcess(), ctypes.byref(counters), counters.cb):
        raise OSError('GetProcessMemoryInfo failed')
    return int(counters.PeakWorkingSetSize)


def interval(values):
    return [float(v) for v in np.quantile(values, [0.025, 0.975], method='linear')]


def distribution(values, point):
    return {'point': float(point), 'bootstrap_mean': float(np.mean(values)),
            'bootstrap_median': float(np.median(values)), 'ci95': interval(values),
            'fraction_positive': float(np.mean(values > 0)),
            'fraction_nonnegative': float(np.mean(values >= 0))}


def binomial_cdf(k, n, probability):
    return sum(math.comb(n, j) * probability**j * (1 - probability)**(n - j) for j in range(k + 1))


def exact_binomial_interval(k, n):
    """Two-sided Clopper-Pearson via deterministic bisection; no scipy needed."""
    if k == 0:
        lower = 0.0
    else:
        left, right = 0.0, 1.0
        for _ in range(100):
            middle = (left + right) / 2
            survival = 1 - binomial_cdf(k - 1, n, middle)
            if survival < .025:
                left = middle
            else:
                right = middle
        lower = (left + right) / 2
    if k == n:
        upper = 1.0
    else:
        left, right = 0.0, 1.0
        for _ in range(100):
            middle = (left + right) / 2
            if binomial_cdf(k, n, middle) > .025:
                left = middle
            else:
                right = middle
        upper = (left + right) / 2
    return [lower, upper]

def resample_operations(cal, normal, bad_supported, bad_any, replicates, seed):
    n_cal, n_normal, n_bad = len(cal), len(normal), len(bad_any)
    order = math.ceil(.95 * (n_cal + 1)) - 1
    if order >= n_cal:
        raise ValueError('Synthetic/general helper requires an observed calibration rank')
    tau = np.partition(cal, order, axis=0)[order]
    point_alarms = (normal > tau, bad_supported > tau, bad_any > tau)
    point = np.asarray([a.mean(axis=0) for a in point_alarms]).T
    rng = np.random.Generator(np.random.PCG64(seed))
    cal_draws = rng.integers(0, n_cal, size=(replicates, n_cal), dtype=np.int64)
    normal_draws = rng.integers(0, n_normal, size=(replicates, n_normal), dtype=np.int64)
    bad_draws = rng.integers(0, n_bad, size=(replicates, n_bad), dtype=np.int64)
    tau_draws = np.partition(cal[cal_draws], order, axis=1)[:, order, :]
    bootstrap = {'fixed_calibration': [], 'calibration_only': [], 'joint': []}
    for scores, draws in ((normal, normal_draws), (bad_supported, bad_draws), (bad_any, bad_draws)):
        bootstrap['fixed_calibration'].append((scores[draws] > tau).mean(axis=1))
        bootstrap['calibration_only'].append((scores[None, :, :] > tau_draws[:, None, :]).mean(axis=1))
        bootstrap['joint'].append((scores[draws] > tau_draws[:, None, :]).mean(axis=1))
    bootstrap = {regime: np.stack(values, axis=2) for regime, values in bootstrap.items()}
    return tau, point_alarms, point, tau_draws, bootstrap, (cal_draws, normal_draws, bad_draws)


def analyze(plan, freeze, identities, groups, spatial):
    labels = official_labels(identities)
    cal_keys = sorted(g for g, rows in groups.items() if rows[0]['research_role'] == 'source_calibration')
    dev_keys = sorted(g for g, rows in groups.items() if rows[0]['research_role'] == 'development')
    normal_keys = [g for g in dev_keys if not any(labels[r['path']] for r in groups[g])]
    abnormal_keys = [g for g in dev_keys if any(labels[r['path']] for r in groups[g])]
    check('group_strata', (len(cal_keys), len(normal_keys), len(abnormal_keys)) == (76, 44, 107))
    check('rank74', math.ceil(.95 * (len(cal_keys) + 1)) == 74)

    def group_max(keys, supported=False):
        return np.asarray([[max(float(r[m]) for r in groups[g] if not supported or labels[r['path']])
                            for m in METHODS] for g in keys], dtype=np.float64)
    cal = group_max(cal_keys)
    normal = group_max(normal_keys)
    bad_supported = group_max(abnormal_keys, True)
    bad_any = group_max(abnormal_keys)
    replicates = plan['uncertainty']['bootstrap_replicates']
    tau, point_alarms, point, tau_draws, bootstrap, draws = resample_operations(
        cal, normal, bad_supported, bad_any, replicates, plan['uncertainty']['seed'])
    cal_draws, normal_draws, bad_draws = draws

    operating_rows, group_rows, cal_rows, bootstrap_methods = [], [], [], []
    for j, method in enumerate(METHODS):
        fp, tp, atp = (int(a[:, j].sum()) for a in point_alarms)
        cp = exact_binomial_interval(fp, 44)
        operating_rows.append({'method': method, 'group_threshold': float(tau[j]), 'normal_groups': 44,
                               'abnormal_groups': 107, 'normal_false_positive_groups': fp,
                               'supported_detected_groups': tp, 'any_detected_groups': atp,
                               'normal_group_fpr': fp / 44, 'supported_group_tpr': tp / 107,
                               'any_group_tpr': atp / 107, 'normal_fpr_cp95_lower': cp[0],
                               'normal_fpr_cp95_upper': cp[1]})
        for keys, scores, is_bad in ((normal_keys, normal, False), (abnormal_keys, bad_any, True)):
            for i, key in enumerate(keys):
                supported_score = float(bad_supported[i, j]) if is_bad else None
                group_rows.append({'method': method, 'group_key': key, 'is_anomaly': int(is_bad),
                                   'group_score': float(scores[i, j]), 'supported_score': supported_score,
                                   'threshold': float(tau[j]), 'any_alarm': int(scores[i, j] > tau[j]),
                                   'supported_alarm': int(is_bad and supported_score > tau[j])})
        for i, key in enumerate(cal_keys):
            cal_rows.append({'method': method, 'group_key': key, 'group_score': float(cal[i, j])})
        for regime, values in bootstrap.items():
            for k, metric in enumerate(METRICS):
                bootstrap_methods.append({'method': method, 'regime': regime, 'metric': metric,
                                          **distribution(values[:, j, k], point[j, k])})

    contrast_rows, contrast_bootstrap = [], []
    for candidate, reference in PAIRS:
        candidate_ids = [METHODS.index(f'{candidate}_s{s}') for s in range(3)]
        reference_ids = [METHODS.index(f'{reference}_s{s}') for s in range(3)]
        for seed, j, k in zip(range(3), candidate_ids, reference_ids):
            candidate_only = int(np.sum(point_alarms[0][:, j] & ~point_alarms[0][:, k]))
            reference_only = int(np.sum(~point_alarms[0][:, j] & point_alarms[0][:, k]))
            contrast_rows.append({'candidate': candidate, 'reference': reference, 'seed': seed,
                                  'delta_fpr': float(point[j, 0] - point[k, 0]),
                                  'delta_supported_tpr': float(point[j, 1] - point[k, 1]),
                                  'delta_any_tpr': float(point[j, 2] - point[k, 2]),
                                  'normal_candidate_only_alarm': candidate_only,
                                  'normal_reference_only_alarm': reference_only})
        for regime, values in bootstrap.items():
            delta = values[:, candidate_ids, :] - values[:, reference_ids, :]
            delta_point = point[candidate_ids, :] - point[reference_ids, :]
            for seed in (*range(3), 'fixed_seed_mean'):
                draw = delta.mean(axis=1) if seed == 'fixed_seed_mean' else delta[:, seed, :]
                actual = delta_point.mean(axis=0) if seed == 'fixed_seed_mean' else delta_point[seed, :]
                for q, metric in enumerate(METRICS):
                    contrast_bootstrap.append({'candidate': candidate, 'reference': reference, 'seed': seed,
                                               'regime': regime, 'metric': metric,
                                               **distribution(draw[:, q], actual[q])})

    normal_contrasts, coverage_rows, camera_rows, peak_rows = [], [], [], []
    g_primary = {}
    for candidate, reference in PAIRS:
        for seed in range(3):
            cand, ref = f'{candidate}_s{seed}', f'{reference}_s{seed}'
            # Reference/candidate sign: positive means reference normal scores are higher.
            readout_vectors = {}
            for readout in READOUTS:
                field = 'max' if readout == 'full' else readout + '_max'
                vector = np.asarray([np.mean([math.log(float(spatial[(ref, r['path'])][field]) /
                                                               float(spatial[(cand, r['path'])][field]))
                                                   for r in groups[g]]) for g in cal_keys])
                readout_vectors[readout] = vector
                normal_contrasts.append({'candidate': candidate, 'reference': reference, 'seed': seed,
                                         'endpoint': f'log_{reference}_over_{candidate}:{readout}',
                                         **distribution(vector[cal_draws].mean(axis=1), vector.mean())})
                if candidate == 'balanced' and reference == 'odd' and readout == 'full':
                    g_primary[seed] = float(vector.mean())
            for readout in READOUTS[1:]:
                vector = readout_vectors['full'] - readout_vectors[readout]
                normal_contrasts.append({'candidate': candidate, 'reference': reference, 'seed': seed,
                                         'endpoint': f'full_minus_{readout}',
                                         **distribution(vector[cal_draws].mean(axis=1), vector.mean())})
            j, k = METHODS.index(cand), METHODS.index(ref)
            normal_contrasts.append({'candidate': candidate, 'reference': reference, 'seed': seed,
                                     'endpoint': f'log_tau_{reference}_over_{candidate}',
                                     **distribution(np.log(tau_draws[:, k] / tau_draws[:, j]), math.log(tau[k] / tau[j]))})
    for method in METHODS:
        records = [spatial[(method, r['path'])] for g in cal_keys for r in groups[g]]
        for field in ('mean', 'p95', 'max'):
            values = np.asarray([float(r[field]) for r in records])
            coverage_rows.append({'method': method, 'population': 'source_calibration', 'field': field,
                                  'views': 380, 'mean': float(values.mean()), 'median': float(np.median(values)),
                                  'q95_across_views': float(np.quantile(values, .95)), 'max_across_views': float(values.max())})
        for camera in range(1, 6):
            camera_records = [r for r in records if int(r['camera']) == camera]
            for readout in READOUTS:
                field = 'max' if readout == 'full' else readout + '_max'
                values = np.asarray([float(r[field]) for r in camera_records])
                camera_rows.append({'method': method, 'camera': camera, 'readout': readout,
                                    'mean': float(values.mean()), 'median': float(np.median(values)),
                                    'q95': float(np.quantile(values, .95)), 'max': float(values.max())})
        for population, keys in (('source_calibration', cal_keys), ('development_normal', normal_keys)):
            records = [spatial[(method, r['path'])] for g in keys for r in groups[g]]
            boundary = sum(int(r['argmax_row']) in (0, 63) or int(r['argmax_col']) in (0, 63) for r in records)
            peak_rows.append({'method': method, 'population': population, 'views': len(records),
                              'representative_boundary_peaks': boundary,
                              'unique_peaks': sum(int(r['image_max_tie_count']) == 1 for r in records),
                              **{f'representative_parity{p}': sum(str(int(r['argmax_row']) % 2) + str(int(r['argmax_col']) % 2) == p for r in records)
                                 for p in PHASES}})

    main_contrasts = [r for r in contrast_rows if r['candidate'] == 'balanced' and r['reference'] == 'odd']
    main_joint = next(r for r in contrast_bootstrap if r['candidate'] == 'balanced' and r['reference'] == 'odd'
                      and r['seed'] == 'fixed_seed_mean' and r['regime'] == 'joint' and r['metric'] == 'supported_group_tpr')
    new_seed_gates = {str(s): {'delta_supported_tpr': main_contrasts[s]['delta_supported_tpr'],
                              'normal_mean_log_odd_over_balanced': g_primary[s],
                              'pass': bool(main_contrasts[s]['delta_supported_tpr'] >= .10 and g_primary[s] > 0)}
                      for s in (1, 2)}
    random_rows = [r for r in contrast_rows if r['candidate'] == 'random' and r['reference'] == 'balanced']
    random_screen = all(r['delta_supported_tpr'] >= -.05 and
                        round(r['delta_fpr'] * 44) <= 1 for r in random_rows)
    scientific_gate = all(r['pass'] for r in new_seed_gates.values()) and main_joint['ci95'][0] > 0
    thresholds = [{'method': m, **distribution(tau_draws[:, j], tau[j]),
                   'distinct_resampled_thresholds': int(len(np.unique(tau_draws[:, j])))} for j, m in enumerate(METHODS)]
    draws_receipt = {'generator': 'numpy.random.Generator(PCG64(20261031))',
                     'allocation_order': ['calibration (5000,76)', 'normal development (5000,44)', 'abnormal development (5000,107)'],
                     'dtype': 'int64', 'indices_hash_endianness': sys.byteorder,
                     'sha256_raw_contiguous': {name: hashlib.sha256(draw.tobytes(order='C')).hexdigest()
                                               for name, draw in (('calibration', cal_draws), ('normal_development', normal_draws), ('abnormal_development', bad_draws))}}
    return {'operating_rows': operating_rows, 'group_rows': group_rows, 'calibration_rows': cal_rows,
            'contrast_rows': contrast_rows, 'bootstrap_methods': bootstrap_methods,

def self_test():
    """Synthetic statistical checks only. Does not open research inputs."""
    def two_columns(values):
        values = np.asarray(values, dtype=np.float64)
        return np.column_stack([values, 2 * values])
    cal = two_columns(np.arange(1, 77))
    normal = two_columns(np.arange(51, 95))
    support = two_columns(np.arange(21, 128))
    any_score = two_columns(np.arange(31, 138))
    tau, alarms, point, tau_b, boot, draws = resample_operations(cal, normal, support, any_score, 64, 91817)
    check('selftest_order_statistic_not_interpolated', np.array_equal(tau, [74, 148]))
    check('selftest_strict_greater', alarms[0][23].tolist() == [False, False] and alarms[0][24].tolist() == [True, True])
    check('selftest_supported_is_separate', np.all(point[:, 1] < point[:, 2]))
    check('selftest_scaled_methods_have_same_point_alarms', np.array_equal(point[0], point[1]))
    check('selftest_calibration_variation_present', len(np.unique(tau_b[:, 0])) > 1)
    check('selftest_pairing_preserved', all(np.array_equal(values[:, 0], values[:, 1]) for values in boot.values()))
    for b in range(64):
        reference_taus = np.asarray([sorted(cal[draws[0][b], j])[73] for j in range(2)])
        check('selftest_resampled_rank:' + str(b), np.array_equal(reference_taus, tau_b[b]))
        for q, (scores, indices) in enumerate(((normal, draws[1]), (support, draws[2]), (any_score, draws[2]))):
            check('selftest_joint_manual:' + str(b) + ':' + str(q),
                  np.array_equal(np.mean(scores[indices[b]] > reference_taus, axis=0), boot['joint'][b, :, q]))
    cp0 = exact_binomial_interval(0, 44)
    cpall = exact_binomial_interval(44, 44)
    check('selftest_zero_binomial_bound', abs(cp0[1] - (1 - .025**(1 / 44))) < 1e-12 and cp0[0] == 0)
    check('selftest_all_binomial_bound', abs(cpall[0] - .025**(1 / 44)) < 1e-12 and cpall[1] == 1)
    check('selftest_complement_binomial', max(abs(a - b) for a, b in zip(exact_binomial_interval(1, 44),
          [1 - exact_binomial_interval(43, 44)[1], 1 - exact_binomial_interval(43, 44)[0]])) < 1e-12)
    result = {'status': 'PASS_SYNTHETIC_ONLY', 'checks': len(CHECKS), 'failed': 0,
              'analysis_sha256': sha256(Path(__file__)), 'numpy_version': np.__version__,
              'wall_seconds': time.perf_counter() - START, 'cpu_seconds': time.process_time() - CPU_START,
              'peak_rss_bytes': peak_rss_bytes(), 'research_inputs_read': 0, 'gpu': 0, 'network': 0}
    write_json('analysis_selftest.json', result)
    print(json.dumps(result))
