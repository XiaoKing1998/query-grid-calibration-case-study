def digest(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(8*1024*1024), b''):
            h.update(block)
    return h.hexdigest()


def data(path):
    INPUT_HASHES[str(path)] = digest(path)
    return json.loads(Path(path).read_text(encoding='utf-8-sig'))


def table(path):
    INPUT_HASHES[str(path)] = digest(path)
    with Path(path).open(encoding='utf-8-sig', newline='') as stream:
        return list(csv.DictReader(stream))


def check(name, condition):
    CHECKS.append(name)
    if not bool(condition):
        raise AssertionError(name)


def close(name, observed, expected, atol=2e-11):
    global MAX_ERROR
    a, b = np.asarray(observed, dtype=float), np.asarray(expected, dtype=float)
    check(name+':shape', a.shape == b.shape)
    error = float(np.max(np.abs(a-b))) if a.size else 0.0
    MAX_ERROR = max(MAX_ERROR, error)
    check(name, np.isfinite(a).all() and np.isfinite(b).all() and np.allclose(a, b, rtol=0, atol=atol))


def peak_rss():
    class Mem(ctypes.Structure):
        _fields_ = [('cb', ctypes.c_ulong), ('PageFaultCount', ctypes.c_ulong),
                    ('PeakWorkingSetSize', ctypes.c_size_t), ('WorkingSetSize', ctypes.c_size_t),
                    ('QuotaPeakPagedPoolUsage', ctypes.c_size_t), ('QuotaPagedPoolUsage', ctypes.c_size_t),
                    ('QuotaPeakNonPagedPoolUsage', ctypes.c_size_t), ('QuotaNonPagedPoolUsage', ctypes.c_size_t),
                    ('PagefileUsage', ctypes.c_size_t), ('PeakPagefileUsage', ctypes.c_size_t)]
    k = ctypes.WinDLL('kernel32'); k.GetCurrentProcess.restype = ctypes.c_void_p
    p = ctypes.WinDLL('psapi'); p.GetProcessMemoryInfo.argtypes = [ctypes.c_void_p, ctypes.c_void_p, ctypes.c_ulong]
    v = Mem(); v.cb = ctypes.sizeof(v)
    if not p.GetProcessMemoryInfo(k.GetCurrentProcess(), ctypes.byref(v), v.cb):
        raise OSError('GetProcessMemoryInfo')
    return int(v.PeakWorkingSetSize)


def check_distribution(name, row, samples, point):
    expected = {'point': point, 'bootstrap_mean': np.mean(samples), 'bootstrap_median': np.median(samples),
                'ci95': np.quantile(samples, [.025, .975]), 'fraction_positive': np.mean(samples > 0),
                'fraction_nonnegative': np.mean(samples >= 0)}
    for field, value in expected.items():
        close(name+':'+field, row[field], value)


def unique_index(records, fields):
    result = {tuple(row[k] for k in fields): row for row in records}
    check('unique:'+':'.join(fields), len(result) == len(records))
    return result


def candidate_check(preflight, inputs):

    check('candidate_fit_count', len(fit) == len(frozen['identities']) == 875)
    coord = {c: [] for c in ('odd', 'balanced', 'random')}
    order_hash = hashlib.sha256()
    for i, (source, saved) in enumerate(zip(fit, frozen['identities'])):
        check('candidate_metadata:'+str(i), all(source[k] == saved[k] for k in IDENTITY) and saved['fit_image_index'] == i)
        canonical = {k: source[k] for k in ('path', 'group_key', 'category', 'camera', 'research_role')}
        canonical['official_split'] = 'train'
        order_hash.update((json.dumps(canonical, sort_keys=True, ensure_ascii=False, separators=(',', ':'))+'\n').encode())
        path = source['path']
        seeds = [int.from_bytes(hashlib.sha256((tag+path).encode()).digest()[:8], 'little')
                 for tag in ('bank256:', 'phase256:', 'random256-v1:')]
        coarse = np.random.Generator(np.random.PCG64(seeds[0])).choice(1024, size=256, replace=False)
        parity = np.random.Generator(np.random.PCG64(seeds[1])).permutation(np.tile(np.arange(4), 64))
        row, col = np.divmod(coarse, 32)
        expected = {'odd': 64*(2*row+1)+2*col+1,
                    'balanced': 64*(2*row+parity//2)+2*col+parity%2,
                    'random': np.random.Generator(np.random.PCG64(seeds[2])).choice(4096, size=256, replace=False)}
        for construction, arr in expected.items():
            check('candidate_members_order:'+construction+':'+str(i), np.array_equal(saved[construction], arr))
            check('candidate_unique_bounds:'+construction+':'+str(i), len(set(arr)) == 256 and min(arr) >= 0 and max(arr) < 4096)
            coord[construction].append(arr)
    check('source_order_hash', order_hash.hexdigest() == preflight['source_order_sha256'] == frozen['source_order_sha256'])
    for construction, rows in coord.items():
        sha = hashlib.sha256(np.array(rows, dtype='<i8').tobytes()).hexdigest()
        check('candidate_coordinate_hash:'+construction, sha == frozen['candidate_coordinate_sha256'][construction] == preflight['candidate_coordinate_sha256'][construction])
    return fit, coord


def regions():
    result = {'top_left': (127, lambda r,c: r == 0 or c == 0),
              'bottom_right': (127, lambda r,c: r == 63 or c == 63),
              'outer_ring': (252, lambda r,c: r in (0,63) or c in (0,63)),
              'interior': (3844, lambda r,c: 1 <= r <= 62 and 1 <= c <= 62)}
    for p in PHASES:
        a,b = map(int,p)
        result['parity'+p] = (1024, lambda r,c,a=a,b=b: r%2 == a and c%2 == b)
        result['interior_parity'+p] = (961, lambda r,c,a=a,b=b: 1 <= r <= 62 and 1 <= c <= 62 and r%2 == a and c%2 == b)
    for name,(count,predicate) in result.items():
        check('independent_region_count:'+name, sum(predicate(r,c) for r in range(64) for c in range(64)) == count)
    return result
