def sha(path):
    value = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(8*1024*1024), b''):
            value.update(block)
    return value.hexdigest()


def read(path):
    INPUT_SHA[str(path)] = sha(path)
    return json.loads(Path(path).read_text(encoding='utf-8-sig'))


def rows(path):
    INPUT_SHA[str(path)] = sha(path)
    with Path(path).open(encoding='utf-8-sig', newline='') as stream:
        return list(csv.DictReader(stream))


def check(name, ok):
    CHECKS.append(name)
    if not bool(ok):
        raise AssertionError(name)


def numeric(name, actual, expected, tolerance=2e-10):
    global MAX_ERROR
    a, b = np.asarray(actual, dtype=float), np.asarray(expected, dtype=float)
    check(name+':shape', a.shape == b.shape)
    check(name+':finite', np.isfinite(a).all() and np.isfinite(b).all())
    error = float(np.abs(a-b).max()) if a.size else 0.0
    MAX_ERROR = max(MAX_ERROR, error)
    check(name, np.allclose(a,b,rtol=0,atol=tolerance))


def ratio_values(pair):
    values = {name: float(pair[name]) for name in ('t_norm2','c_norm2','t_b2','c_b2','t_c2','t_residual2','c_residual2')}
    if not all(math.isfinite(value) and value >= 0 for value in values.values()):
        return None
    replace_mean = (values['t_b2']+values['c_b2'])/2
    residual_mean = (values['t_residual2']+values['c_residual2'])/2
    direct_mean = (math.sqrt(values['t_b2'])+math.sqrt(values['c_b2']))/2
    if min(replace_mean,residual_mean,direct_mean) <= 0:
        return None
    replace = abs(values['t_b2']-values['c_b2'])/replace_mean
    residual = abs(values['t_residual2']-values['c_residual2'])/residual_mean
    direct = math.sqrt(values['t_c2'])/direct_mean
    norm_mean = (values['t_norm2']+values['c_norm2'])/2
    return {'replace_relative':replace, 'residual_relative':residual, 'direct_ratio':direct,
            'norm_relative':abs(values['t_norm2']-values['c_norm2'])/norm_mean if norm_mean > 0 else None,
            'cost':replace/.10+residual/.10+direct/.5,
            'legal':replace <= .10 and residual <= .10 and direct <= .5}


def audit_pair(pair, candidates, anchor_by_id, require_legal=False):
    """Recompute formulas and membership, not the unsaved feature distances."""
    anchor = anchor_by_id[int(pair['anchor_id'])]
    path = pair['path']; saved = candidates[path]
    name = path+':'+str(pair['anchor_id'])+':'+pair['target_phase']+':'+str(pair['sham_cell'])
    check('pair_source:'+name, path == anchor['path'] and pair['group_key'] == saved['group_key'] == anchor['group_key'] and str(pair['camera']) == str(saved['camera']) == str(anchor['camera']))
    check('pair_fit_role:'+name,pair['research_role'] == saved['research_role'] == 'source_fit' and pair['category'] == saved['category'])
    r,c = int(anchor['row']),int(anchor['col'])
    check('eligible_anchor:'+name, r%2 == c%2 == 1 and 3 <= r <= 61 and 3 <= c <= 61)
    check('anchor_coordinates:'+name, int(pair['anchor_row']) == r and int(pair['anchor_col']) == c)
    phase = pair['target_phase']; check('target_phase:'+name,phase in PHASES)
    target = 64*(r-1+int(phase[0]))+c-1+int(phase[1])
    check('target_identity:'+name,int(pair['target_cell']) == target)
    check('target_coordinate_export:'+name,64*int(pair['target_row'])+int(pair['target_col']) == target)
    sham = int(pair['sham_cell']); sr,sc = divmod(sham,64)
    check('sham_coordinate_export:'+name,int(pair['sham_row']) == sr and int(pair['sham_col']) == sc)
    check('sham_membership:'+name,sham in saved['odd'] and 1 <= sr <= 62 and 1 <= sc <= 62 and sr%2 == sc%2 == 1)
    check('sham_new_identity:'+name,sham not in saved['b0_cells'])
    ratios = ratio_values(pair)
    check('pair_positive_denominators:'+name, ratios is not None)
    for key,value in ratios.items():
        if key == 'legal':continue
        if value is not None:numeric('pair_formula:'+name+':'+key,pair[key],value)
    check('pair_legal_flag:'+name,str(pair['all_hard_gates']).lower() == str(ratios['legal']).lower())
    if require_legal:check('pair_three_gates:'+name,ratios['legal'])
    return ratios

def peak_rss():
    class Mem(ctypes.Structure):
        _fields_ = [('cb',ctypes.c_ulong),('PageFaultCount',ctypes.c_ulong),
                    ('PeakWorkingSetSize',ctypes.c_size_t),('WorkingSetSize',ctypes.c_size_t),
                    ('QuotaPeakPagedPoolUsage',ctypes.c_size_t),('QuotaPagedPoolUsage',ctypes.c_size_t),
                    ('QuotaPeakNonPagedPoolUsage',ctypes.c_size_t),('QuotaNonPagedPoolUsage',ctypes.c_size_t),
                    ('PagefileUsage',ctypes.c_size_t),('PeakPagefileUsage',ctypes.c_size_t)]
    kernel=ctypes.WinDLL('kernel32');kernel.GetCurrentProcess.restype=ctypes.c_void_p
    psapi=ctypes.WinDLL('psapi');psapi.GetProcessMemoryInfo.argtypes=[ctypes.c_void_p,ctypes.c_void_p,ctypes.c_ulong]
    value=Mem();value.cb=ctypes.sizeof(value)
    if not psapi.GetProcessMemoryInfo(kernel.GetCurrentProcess(),ctypes.byref(value),value.cb):
        raise OSError('GetProcessMemoryInfo')
    return int(value.PeakWorkingSetSize)


def solve_exported_graph(costs, sentinel=1000.0):
    """Full linear assignment; transposed relative to source-major producer."""
    matrix=np.asarray(costs,dtype=float)
    check('assignment_matrix_shape',matrix.ndim == 2 and matrix.shape[1] == 48)
    check('assignment_cost_finite',np.isfinite(matrix).all())
    # Slots are rows here. A full solution must assign all48 slots to distinct sources.
    slot,source=linear_sum_assignment(matrix.T)
    legal=(matrix[source,slot] < sentinel)
    feasible=len(slot) == 48 and bool(legal.all())
    return {'feasible':feasible,'assigned_slots':len(slot),'legal_slots':int(legal.sum()),
            'sentinel_slots':int((~legal).sum()),'total_cost':float(matrix[source,slot].sum()),
            'pairs':[(int(i),int(j)) for i,j in zip(source,slot)]}


def phase_hall_certificate(costs,sentinel=1000.0):
    """Exact capacity test for three phase types, each repeated sixteen times.

    Because all sixteen slots of a phase have identical neighbors, Hall's
    condition reduces to seven nonempty subsets of the three phase types.
    This independently checks feasibility without depending on LSA behavior.
    """
    matrix=np.asarray(costs,dtype=float)
    check('hall_matrix_shape',matrix.ndim == 2 and matrix.shape[1] == 48)
    for index,phase in enumerate(PHASES):
        block=matrix[:,16*index:16*(index+1)]
        check('hall_identical_phase_slots:'+phase,np.array_equal(block,np.repeat(block[:,:1],16,axis=1)))
    adjacency=matrix[:,[0,16,32]] < sentinel
    subsets=[]
    for mask in range(1,8):
        indexes=[i for i in range(3) if mask&(1<<i)]
        union=int(np.any(adjacency[:,indexes],axis=1).sum())
        required=16*len(indexes)
        subsets.append({'phases':[PHASES[i] for i in indexes],'distinct_source_neighbors':union,
                        'required_sources':required,'slack':union-required,'passes':union>=required})
    return {'feasible':all(row['passes'] for row in subsets),'subsets':subsets}

    for metric,tol in [('mean',.005),('p95',.01),('p99',.01)]:
        residual=abs(summary['target'][metric]-summary['sham'][metric])/summary['base'][metric]
        numeric('coverage_residual:'+metric,audit['residuals'][metric]['absolute_difference_relative_to_base'],residual)
        check('coverage_gate:'+metric,audit['residuals'][metric]['tolerance'] == tol and audit['residuals'][metric]['pass'] == (residual <= tol))
        gates[metric]=residual <= tol
        for method in ('target','sham'):numeric('coverage_change_descriptive:'+method+metric,audit['changes_relative_to_base_descriptive'][metric][method],summary[method][metric]/summary['base'][metric]-1)
    check('coverage_decision',audit['status'] == ('FIT_COVERAGE_MATCH_PASS' if all(gates.values()) else 'MATCHING_INCONCLUSIVE'))
    check('coverage_scope',audit['improvement_is_required'] is False and audit['source_group_exclusion'] is True and audit['query_pixels_requested'] == 0)
    group_residuals={g:{metric:(aggregate['target',g][metric]-aggregate['sham',g][metric])/aggregate['base',g][metric] for metric in ('mean','p95','p99')} for m,g in aggregate if m == 'base'}
    camera_summary={method:{camera:{metric:float(np.mean([float(r[metric]) for r in cameras[method,camera]])) for metric in ('mean','p95','p99')} for camera in '12345'} for method in ('base','target','sham')}
    return {'pass':all(gates.values()),'gates':gates,'summary':summary,'camera_summary':camera_summary,'group_residuals':group_residuals}
