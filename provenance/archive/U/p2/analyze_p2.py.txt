def read(path):
    return json.loads(path.read_text(encoding='utf-8-sig'))


def csvrows(path):
    with path.open(encoding='utf-8-sig', newline='') as stream:
        return list(csv.DictReader(stream))


def sha(path):
    result = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(2**20), b''):
            result.update(block)
    return result.hexdigest()


def check(name, value):

def peak_rss():
    class Counters(ctypes.Structure):
        _fields_ = [('cb', ctypes.c_ulong), ('faults', ctypes.c_ulong)] + [(name, ctypes.c_size_t) for name in
            ('PeakWorkingSetSize', 'WorkingSetSize', 'QuotaPeakPagedPoolUsage', 'QuotaPagedPoolUsage',
             'QuotaPeakNonPagedPoolUsage', 'QuotaNonPagedPoolUsage', 'PagefileUsage', 'PeakPagefileUsage')]
    kernel, psapi = ctypes.WinDLL('kernel32'), ctypes.WinDLL('psapi')
    kernel.GetCurrentProcess.restype = ctypes.c_void_p
    psapi.GetProcessMemoryInfo.argtypes = [ctypes.c_void_p, ctypes.c_void_p, ctypes.c_ulong]
    counters = Counters()
    counters.cb = ctypes.sizeof(counters)
    if not psapi.GetProcessMemoryInfo(kernel.GetCurrentProcess(), ctypes.byref(counters), counters.cb):
        raise OSError('memory query failed')
    return int(counters.PeakWorkingSetSize)


def interval(values):
    return [float(v) for v in np.quantile(values, [.025, .975], method='linear')]


def summarize_draws(values, point):
    return {'point': float(point), 'ci95': interval(values), 'bootstrap_mean': float(values.mean()),
            'fraction_positive': float(np.mean(values > 0))}


def binomial_interval(k, n):
    def cdf(j, p):
        return sum(math.comb(n, i) * p**i * (1 - p)**(n - i) for i in range(j + 1))
    if k == 0:
        lower = 0.
    else:
        left, right = 0., 1.
        for _ in range(100):
            mid = (left + right) / 2
            if 1 - cdf(k - 1, mid) < .025:
                left = mid
            else:
                right = mid
        lower = (left + right) / 2
    if k == n:
        upper = 1.
    else:
        left, right = 0., 1.
        for _ in range(100):
            mid = (left + right) / 2
            if cdf(k, mid) > .025:
                left = mid
            else:
                right = mid
        upper = (left + right) / 2
    return [lower, upper]


def region_analysis(vectors, plan):
    count, seed = plan['bootstrap']['region_draws'], plan['bootstrap']['region_seed']
    draw = np.random.Generator(np.random.PCG64(seed)).integers(0, len(vectors['full']),
            size=(count, len(vectors['full'])), dtype=np.int64)
    values = dict(vectors)
    values['non11_mean'] = sum(values['interior_parity' + phase] for phase in ('00', '01', '10')) / 3
    values['interaction_non11_minus11'] = values['non11_mean'] - values['interior_parity11']
    results = {name: summarize_draws(vector[draw].mean(axis=1), vector.mean()) for name, vector in values.items()}
    rule = plan['regional_prediction']
    flags = {
        'each_non11_positive': all(results['interior_parity' + p]['point'] > 0 for p in ('00', '01', '10')),
        'non11_mean_min': results['non11_mean']['point'] >= rule['non11_mean_min'],
        'interaction_min': results['interaction_non11_minus11']['point'] >= rule['interaction_min'],
        'interaction_ci_lower': results['interaction_non11_minus11']['ci95'][0] > rule['interaction_ci95_lower_gt'],
        'abs_g11_max': abs(results['interior_parity11']['point']) <= rule['abs_g11_max'],
        'g11_ci_inside': results['interior_parity11']['ci95'][0] >= rule['g11_ci95_inside'][0] and
                        results['interior_parity11']['ci95'][1] <= rule['g11_ci95_inside'][1],
        'full_g_positive': results['full']['point'] > rule['full_g_gt'],
    }
    return results, {key: bool(value) for key, value in flags.items()}, draw


def operation_analysis(cal, normal, supported, any_bad, plan):
    rank = math.ceil(.95 * (len(cal) + 1))
    if rank > len(cal):
        raise ValueError('Unavailable observed calibration rank')
    tau = np.partition(cal, rank - 1, axis=0)[rank - 1]
    alarms = (normal > tau, supported > tau, any_bad > tau)
    point = np.asarray([v.mean(axis=0) for v in alarms]).T
    rng = np.random.Generator(np.random.PCG64(plan['bootstrap']['operation_seed']))
    count = plan['bootstrap']['operation_draws']
    cdraw = rng.integers(0, len(cal), size=(count, len(cal)), dtype=np.int64)
    ndraw = rng.integers(0, len(normal), size=(count, len(normal)), dtype=np.int64)
    adraw = rng.integers(0, len(supported), size=(count, len(supported)), dtype=np.int64)
    tau_draw = np.partition(cal[cdraw], rank - 1, axis=1)[:, rank - 1, :]
    distributions = {'fixed_calibration': [], 'calibration_only': [], 'joint': []}
    for scores, draw in ((normal, ndraw), (supported, adraw), (any_bad, adraw)):
        distributions['fixed_calibration'].append((scores[draw] > tau).mean(axis=1))
        distributions['calibration_only'].append((scores[None, :, :] > tau_draw[:, None, :]).mean(axis=1))
        distributions['joint'].append((scores[draw] > tau_draw[:, None, :]).mean(axis=1))
    return tau, alarms, point, tau_draw, {key: np.stack(value, axis=2) for key, value in distributions.items()}, (cdraw, ndraw, adraw)

def main():
    check('outputs_new', not (OUT / 'analysis_result.json').exists())
    plan = read(OUT / 'plan.json')
    identities, groups, spatial, labels = admit(plan)
    cal_keys = sorted(g for g, rs in groups.items() if rs[0]['research_role'] == 'source_calibration')
    dev_keys = sorted(g for g, rs in groups.items() if rs[0]['research_role'] == 'development')
    normal_keys = [g for g in dev_keys if not any(labels[r['path']] for r in groups[g])]
    bad_keys = [g for g in dev_keys if any(labels[r['path']] for r in groups[g])]
    check('counts', (len(cal_keys), len(normal_keys), len(bad_keys)) == (76, 44, 107))
    vectors_by_pair = {}
    for candidate, reference in PAIRS:
        vector = {}
        for readout in READOUTS:
            field = 'max' if readout == 'full' else readout + '_max'
            vector[readout] = np.asarray([np.mean([math.log(float(spatial[(reference, r['path'])][field]) /
                                                                  float(spatial[(candidate, r['path'])][field])) for r in groups[g]])
                                          for g in cal_keys], dtype=np.float64)
        vectors_by_pair[(candidate, reference)] = vector
    regional, flags, regional_draws = region_analysis(vectors_by_pair[('target', 'sham')], plan)
    descriptive = []
    for (candidate, reference), vectors in vectors_by_pair.items():
        for readout, vector in vectors.items():
            descriptive.append({'candidate': candidate, 'reference': reference, 'readout': readout,
                                'sign': 'log(reference/candidate)', **summarize_draws(vector[regional_draws].mean(axis=1), vector.mean())})
    group_regions = []
    for i, key in enumerate(cal_keys):
        values = {name: float(vector[i]) for name, vector in vectors_by_pair[('target', 'sham')].items()}
        values['non11_mean'] = sum(values['interior_parity' + phase] for phase in ('00', '01', '10')) / 3
        values['interaction_non11_minus11'] = values['non11_mean'] - values['interior_parity11']
        group_regions.append({'group_key': key, **values})

    def group_scores(keys, supported_only=False):
        return np.asarray([[max(float(r[m]) for r in groups[g] if not supported_only or labels[r['path']])
                            for m in METHODS] for g in keys], dtype=np.float64)
    cal = group_scores(cal_keys)
    normal, supported, any_bad = group_scores(normal_keys), group_scores(bad_keys, True), group_scores(bad_keys)
    tau, alarms, point, tau_draws, boot, draws = operation_analysis(cal, normal, supported, any_bad, plan)
    operations, bootstrap_methods, group_predictions, calibration_rows = [], [], [], []
    for j, method in enumerate(METHODS):
        fp, tp, any_tp = [int(values[:, j].sum()) for values in alarms]
        cp = binomial_interval(fp, 44)
        operations.append({'method': method, 'threshold': float(tau[j]), 'normal_false_positive_groups': fp,
                           'supported_detected_groups': tp, 'any_detected_groups': any_tp,
                           'normal_group_fpr': fp / 44, 'supported_group_tpr': tp / 107, 'any_group_tpr': any_tp / 107,
                           'normal_fpr_binomial95_lower': cp[0], 'normal_fpr_binomial95_upper': cp[1]})
        for regime, values in boot.items():
            for q, metric in enumerate(METRICS):
                bootstrap_methods.append({'method': method, 'regime': regime, 'metric': metric,
                                          **summarize_draws(values[:, j, q], point[j, q])})
        for i, key in enumerate(cal_keys):
            calibration_rows.append({'method': method, 'group_key': key, 'score': float(cal[i, j])})
        for keys, scores, is_bad in ((normal_keys, normal, False), (bad_keys, any_bad, True)):
            for i, key in enumerate(keys):
                group_predictions.append({'method': method, 'group_key': key, 'is_anomaly': int(is_bad),
                                          'full_group_score': float(scores[i, j]), 'threshold': float(tau[j]),
                                          'any_alarm': int(scores[i, j] > tau[j]),
                                          'supported_alarm': int(is_bad and supported[i, j] > tau[j])})
    comparisons, bootstrap_contrasts = [], []
    for candidate, reference in PAIRS:
        j, k = METHODS.index(candidate), METHODS.index(reference)
        comparisons.append({'candidate': candidate, 'reference': reference,
                            'delta_fpr': float(point[j, 0] - point[k, 0]),
                            'delta_supported_tpr': float(point[j, 1] - point[k, 1]),
                            'delta_any_tpr': float(point[j, 2] - point[k, 2]),
                            'normal_candidate_only_alarm': int(np.sum(alarms[0][:, j] & ~alarms[0][:, k])),
                            'normal_reference_only_alarm': int(np.sum(~alarms[0][:, j] & alarms[0][:, k]))})
        for regime, values in boot.items():
            for q, metric in enumerate(METRICS):
                bootstrap_contrasts.append({'candidate': candidate, 'reference': reference, 'regime': regime,
                                           'metric': metric, **summarize_draws(values[:, j, q] - values[:, k, q], point[j, q] - point[k, q])})
    result = {'status': 'P2_QUERY_ANALYSIS_COMPLETE_PENDING_INDEPENDENT_VERIFICATION',
              'regional_endpoints': regional, 'regional_gate_components': flags,
              'regional_gate_pass': bool(all(flags.values())), 'normal_readout_contrasts': descriptive,
              'operating_points': operations, 'paired_contrasts': comparisons,
              'bootstrap_methods': bootstrap_methods, 'bootstrap_contrasts': bootstrap_contrasts,
              'threshold_distributions': [{'method': method, **summarize_draws(tau_draws[:, j], tau[j]),
                                           'distinct_values': int(len(np.unique(tau_draws[:, j])))} for j, method in enumerate(METHODS)],
              'bootstrap_receipt': {'region_seed': 20261032, 'operation_seed': 20261033, 'draws': 5000,
                                    'operation_allocation_order': ['calibration(5000,76)', 'normal_development(5000,44)', 'abnormal_development(5000,107)'],
                                    'groups_sorted_by': 'group_key', 'dtype': 'int64',
                                    'draw_sha256': {name: hashlib.sha256(value.tobytes()).hexdigest() for name, value in
                                                   zip(('regional', 'operation_calibration', 'operation_normal', 'operation_abnormal'), (regional_draws, *draws))}},
              'decision': {'fit_gates_passed_before_query': True, 'regional_prediction_pass': bool(all(flags.values())),
                           'p3_admission_design_eligible': bool(all(flags.values())), 'confirmation_access_authorized_by_this_analysis': False,
                           'old_three_scientific_gates_remain_failed': True, 'main_contribution_established': False},
              'limitations': ['Fixed-bank repeated development; no independent confirmation', 'Regional gates test specified summaries only, not full feature geometry or positional encoding causation',
                              '44 normal groups cannot establish population equal risk', 'Supported means abnormal-view alarm, not correct defect location',
                              'Bootstrap cannot create unseen normal tails or account for fit/matching choice uncertainty'],
              'resource_receipt': {'wall_seconds_before_output': time.perf_counter() - START, 'cpu_seconds_before_output': time.process_time() - CPU_START,
                                   'peak_rss_bytes': peak_rss(), 'gpu': 0, 'network': 0, 'new_pixels_masks_confirmation_scores': 0,
                                   'command': [sys.executable, '-B', str(Path(__file__).resolve())]}}
    inputs = ['plan.json', 'protocol.md', 'run_p2.py', 'matching_audit.json', 'fit_coverage_audit.json',
              'intervention_freeze.json', 'prediction_freeze.json', 'supervisor.json', 'resource_receipt.json',
              'scalar_predictions.csv', 'spatial_statistics.csv', 'normal_calibration_coverage.csv']
    admission = {'status': 'PASS_P2_QUERY_ANALYSIS_ADMISSION', 'checks': len(CHECKS), 'failed': 0,
                 'input_sha256': {str(OUT / name): sha(OUT / name) for name in inputs},
                 'analysis_sha256': sha(Path(__file__)), 'metadata_sha256': sha(METADATA),
                 'p1_scalar_sha256': sha(P1 / 'scalar_predictions.csv'), 'numpy': np.__version__, 'python': sys.version}
    savecsv('analysis_primary_groups.csv', group_regions)
    savecsv('analysis_operating_points.csv', operations)
    savecsv('analysis_paired_contrasts.csv', comparisons)
    savecsv('analysis_group_predictions.csv', group_predictions)
    savecsv('analysis_calibration_groups.csv', calibration_rows)
    save('analysis_admission.json', admission)
    save('analysis_result.json', result)
    report = ['# P2 条件性query分析', '', '状态：计算完成，等待独立核验。', '',
              '| 正常log(sham/target)端点 | 点估计 | 条件95%区间 |', '|---|---:|---|']
    for name in ('interior_parity00', 'interior_parity01', 'interior_parity10', 'interior_parity11', 'non11_mean', 'interaction_non11_minus11', 'full'):
        value = regional[name]
        report.append(f"| {name} | {value['point']:.8f} | [{value['ci95'][0]:.8f}, {value['ci95'][1]:.8f}] |")
    report += ['', f'全部冻结区域门通过：{all(flags.values())}。', '',
               '| 方法 | group阈值 | FP/44 | supported/107 | any/107 |', '|---|---:|---:|---:|---:|']
    for row in operations:
        report.append(f"| {row['method']} | {row['threshold']:.6f} | {row['normal_false_positive_groups']} | {row['supported_detected_groups']} | {row['any_detected_groups']} |")
    report += ['', '完整base/target/sham工作点、target−sham/target−base/sham−base及fixed/cal-only/joint全部区间见analysis_result.json。',
               '主要操作结果始终使用完整4096格；内部相位各961格仅检验预定机制。',
               '区域bootstrap5000次seed20261032，操作bootstrap5000次seed20261033，采集组配对，不把视图或格点当独立样本。',
               '所有门通过仅允许设计P3准入；失败或不确定结束本主线，不追加第三轮诊断。',
               '即使通过，也不能认定位置编码因果、全部特征几何已匹配、总体等风险、缺陷定位正确或论文主贡献成立。']
    with (OUT / 'analysis_report.md').open('x', encoding='utf-8') as stream:
        stream.write('\n'.join(report) + '\n')
    check('total_saved_budget', sum(p.stat().st_size for folder in (OUT, P1) for p in folder.rglob('*') if p.is_file()) <= plan['resources']['persisted_bytes'])
    save('analysis_completion.json', {'status': 'COMPLETE_PENDING_INDEPENDENT_VERIFICATION',
                                      'result_sha256': sha(OUT / 'analysis_result.json'), 'checks': len(CHECKS), 'failed': 0,
                                      'wall_seconds': time.perf_counter() - START, 'cpu_seconds': time.process_time() - CPU_START,
                                      'peak_rss_bytes': peak_rss()})
    print(json.dumps(result['decision']))


def self_test():
    plan = {'bootstrap': {'region_draws': 64, 'region_seed': 20261032, 'operation_draws': 64, 'operation_seed': 20261033},
            'regional_prediction': {'non11_mean_min': .02, 'interaction_min': .02, 'interaction_ci95_lower_gt': 0,
                                    'abs_g11_max': .01, 'g11_ci95_inside': [-.02, .02], 'full_g_gt': 0}}
    vectors = {'full': np.full(76, .03), **{'interior_parity' + phase: np.full(76, value)
                                        for phase, value in zip(PHASES, (.03, .04, .05, 0.))}}
    estimates, flags, draws = region_analysis(vectors, plan)
    check('selftest_positive_regional_prediction', all(flags.values()))
    check('selftest_equal_phase_average', abs(estimates['non11_mean']['point'] - .04) < 1e-14)
    check('selftest_direct_interaction', abs(estimates['interaction_non11_minus11']['point'] - .04) < 1e-14)
    changed = dict(vectors)
    changed['interior_parity11'] = np.full(76, .03)
